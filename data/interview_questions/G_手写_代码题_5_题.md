import json

def parse_tool_calls(content: str) -> list[dict]:
    text = content.strip()
    if text.startswith("```"):            # 容错：去掉代码块包裹
        text = text.strip("`")
        text = text[text.find("{") : text.rfind("}") + 1]
    try:
        data = json.loads(text)
    except json.JSONDecodeError:
        return []
    return [data] if isinstance(data, dict) else (data if isinstance(data, list) else [])
`

讲点： 模型输出不可信，解析要容错，失败兜底不崩溃。

`
EVAL = [("电机过热怎么排查？", "散热风扇", "retrieve_knowledge"), ...]
def run_eval(agent):
    ok = 0
    for q, key, tool in EVAL:
        events = agent.chat_stream(q, session="eval")
        tools = [e for e in events if e.type == "tool_call"]
        answer = "".join(e for e in events if e.type == "content")
        ok += (tool in tools) and (key.lower() in answer.lower())
    return ok / len(EVAL)   # 综合准确率（工具对 + 答案对）
